cmake_minimum_required(VERSION 3.24)
project(PhononSAPI VERSION 1.0.9 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Phonon SAPI targets Windows")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(VOICE_BITS 64)
else()
  set(VOICE_BITS 32)
endif()
add_library(PhononAdapter SHARED src/engine.cpp src/engine.def)
set_target_properties(PhononAdapter PROPERTIES PREFIX "" OUTPUT_NAME "PhononSAPI${VOICE_BITS}")
target_compile_definitions(PhononAdapter PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(PhononAdapter PRIVATE ole32 uuid sapi advapi32 shell32)
if(MINGW)
  if(NOT SAPI_DDK_INCLUDE)
    message(FATAL_ERROR "Set SAPI_DDK_INCLUDE to the Windows SDK folder containing sapiddk.h")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sdk-headers")
  configure_file("${SAPI_DDK_INCLUDE}/sapiddk.h" "${CMAKE_BINARY_DIR}/sdk-headers/sapiddk.h" COPYONLY)
  target_include_directories(PhononAdapter PRIVATE "${CMAKE_BINARY_DIR}/sdk-headers")
  target_link_options(PhononAdapter PRIVATE -static)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(SONIC_REV b93885dcb70aae50c6f76b0fe4e0868f029a077e)
  set(SONIC_DIR "${CMAKE_BINARY_DIR}/sonic")
  file(MAKE_DIRECTORY "${SONIC_DIR}")
  function(sonic_file NAME HASH)
    set(OUTPUT "${SONIC_DIR}/${NAME}")
    if(NOT EXISTS "${OUTPUT}")
      file(DOWNLOAD "https://raw.githubusercontent.com/waywardgeek/sonic/${SONIC_REV}/${NAME}" "${OUTPUT}"
        EXPECTED_HASH "SHA256=${HASH}" TLS_VERIFY ON)
    endif()
    file(SHA256 "${OUTPUT}" ACTUAL)
    if(NOT ACTUAL STREQUAL HASH)
      message(FATAL_ERROR "Sonic dependency hash mismatch")
    endif()
  endfunction()
  sonic_file(sonic.c 4a21d8086f844e3e68cb4f85961623a69d35da8ca93cf6ea704ccd125e64eaf5)
  sonic_file(sonic.h a2fc087b68c25141e2fb7ab56e71b111bee79cf311fcc850d492eb574c0eeffe)
  add_executable(PhononHost src/host.cpp "${SONIC_DIR}/sonic.c")
  target_include_directories(PhononHost PRIVATE "${SONIC_DIR}")
  target_compile_definitions(PhononHost PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(PhononHost PRIVATE ole32 uuid shell32 advapi32)
  if(MINGW)
    target_link_options(PhononHost PRIVATE -static -municode)
  endif()
  find_program(CSC csc PATHS "$ENV{WINDIR}/Microsoft.NET/Framework64/v4.0.30319" REQUIRED)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/src/Manager.cs" MANAGER_SOURCE)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/src/WindowsInstaller.cs" INSTALLER_SOURCE)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/src/RestartUpdate.cs" RESTART_UPDATE_SOURCE)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/src/CooperativeUpdate.cs" COOPERATIVE_UPDATE_SOURCE)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/src/ImportedVoices.cs" IMPORT_SOURCE)
  file(TO_NATIVE_PATH "${CMAKE_BINARY_DIR}/PhononSAPI.exe" MANAGER_OUTPUT)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/PhononSAPI.exe"
    COMMAND "${CSC}" /nologo /target:winexe /platform:x64 /optimize+
      "/out:${MANAGER_OUTPUT}"
      /r:System.Windows.Forms.dll /r:System.Drawing.dll /r:System.Web.Extensions.dll
      /r:System.Net.Http.dll /r:System.IO.Compression.dll /r:System.IO.Compression.FileSystem.dll /r:Microsoft.CSharp.dll
      "${MANAGER_SOURCE}" "${INSTALLER_SOURCE}" "${RESTART_UPDATE_SOURCE}" "${COOPERATIVE_UPDATE_SOURCE}" "${IMPORT_SOURCE}"
    DEPENDS src/Manager.cs src/WindowsInstaller.cs src/RestartUpdate.cs src/CooperativeUpdate.cs src/ImportedVoices.cs VERBATIM)
  add_custom_target(PhononSettings ALL DEPENDS "${CMAKE_BINARY_DIR}/PhononSAPI.exe")
  configure_file(src/PhononSAPI.exe.config "${CMAKE_BINARY_DIR}/PhononSAPI.exe.config" COPYONLY)
  set(PHONON_SETUP_MSI "" CACHE FILEPATH "MSI to embed in the Phonon setup application")
  set(PHONON_SETUP_MANIFEST "" CACHE FILEPATH "Optional signed update metadata for that MSI")
  if(PHONON_SETUP_MSI)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PHONON_SETUP_MSI}")
    file(SHA256 "${PHONON_SETUP_MSI}" SETUP_HASH)
    file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/Setup.sha256" CONTENT "${SETUP_HASH}")
    set(SETUP_RESOURCES "/resource:${PHONON_SETUP_MSI},PhononSetup.msi"
        "/resource:${CMAKE_BINARY_DIR}/Setup.sha256,PhononSetup.sha256")
    set(SETUP_DEPENDS "${PHONON_SETUP_MSI}" "${CMAKE_BINARY_DIR}/Setup.sha256")
    if(PHONON_SETUP_MANIFEST)
      list(APPEND SETUP_RESOURCES "/resource:${PHONON_SETUP_MANIFEST},PhononSetup.update.json")
      list(APPEND SETUP_DEPENDS "${PHONON_SETUP_MANIFEST}")
    endif()
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/Phonon-SAPI-Setup.exe"
      COMMAND "${CSC}" /nologo /target:winexe /platform:x64 /optimize+ /main:PhononSapi.Setup
        "/out:${CMAKE_BINARY_DIR}/Phonon-SAPI-Setup.exe"
        "/win32manifest:${CMAKE_SOURCE_DIR}/installer/Setup.manifest" ${SETUP_RESOURCES}
        /r:System.Windows.Forms.dll /r:System.Drawing.dll /r:System.Web.Extensions.dll
        /r:System.Net.Http.dll /r:System.IO.Compression.dll /r:System.IO.Compression.FileSystem.dll /r:Microsoft.CSharp.dll
        "${MANAGER_SOURCE}" "${INSTALLER_SOURCE}" "${RESTART_UPDATE_SOURCE}"
        "${COOPERATIVE_UPDATE_SOURCE}" "${IMPORT_SOURCE}" "${CMAKE_SOURCE_DIR}/src/Setup.cs"
      DEPENDS src/Manager.cs src/WindowsInstaller.cs src/RestartUpdate.cs src/CooperativeUpdate.cs src/ImportedVoices.cs
        src/Setup.cs installer/Setup.manifest ${SETUP_DEPENDS} VERBATIM)
    add_custom_target(PhononSetup ALL DEPENDS "${CMAKE_BINARY_DIR}/Phonon-SAPI-Setup.exe")
  endif()
endif()
